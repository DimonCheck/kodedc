✨ Эстетика терминала: fastfetch + bash-алиасы

Превращаем стандартный терминал Astra Linux в современную, информативную и красивую среду.
🚀 fastfetch — быстрая альтернатива neofetch, собранная для совместимости со старыми библиотеками (GLIBC 2.24).

📦 1. Подготовка системы

Устанавливаем компиляторы, cmake, git и wget:

sudo apt update
sudo apt install -y build-essential cmake git wget
Эти пакеты нужны для сборки fastfetch из исходников.
🔧 2. Сборка fastfetch (совместимая версия 2.28.0)

В более новых версиях fastfetch требуется GLIBC ≥ 2.33, поэтому используем проверенную 2.28.0.

cd ~
wget https://github.com/fastfetch-cli/fastfetch/archive/refs/tags/2.28.0.tar.gz
tar -xzf 2.28.0.tar.gz
cd fastfetch-2.28.0
mkdir -p build && cd build
cmake ..
make -j$(nproc)
./fastfetch
✅ Если вы увидели ASCII-логотип Astra Linux — сборка успешна!
⚙️ 3. Создаём конфигурацию (без ошибок)

Конфиг исправляет старый параметр keyWidth → key.width, добавляет отступы и удобный разделитель.

mkdir -p ~/.config/fastfetch
cat > ~/.config/fastfetch/config.jsonc << 'EOF'
{
    "logo": {
        "source": "auto",
        "padding": { "top": 1, "bottom": 1 }
    },
    "display": {
        "separator": "  →  ",
        "key": { "width": 15 }
    },
    "modules": [
        "os", "host", "kernel", "uptime", "packages",
        "shell", "terminal", "cpu", "gpu", "memory", "disk", "localip", "battery"
    ]
}
EOF
💡 Если часть строк обрезается — увеличьте width до 18 или 20.
🎨 4. Алиас, цветные переменные и автозапуск

Добавляем в ~/.bashrc удобную команду fastfetch, цветовые переменные и команду для автоматического запуска.

echo '
# Fastfetch configuration
alias fastfetch="~/fastfetch-2.28.0/build/fastfetch -c ~/.config/fastfetch/config.jsonc"
export FASTFETCH_COLOR_KEYS="yellow"
export FASTFETCH_COLOR_VALUES="green"
export FASTFETCH_COLOR_TITLE="green"
fastfetch
' >> ~/.bashrc
source ~/.bashrc
✨ Теперь при каждом открытии терминала будет запускаться fastfetch с цветами.
🧠 5. Преображение терминала: алиасы, промпт, приветствие

Вставьте этот блок в конец ~/.bashrc (после строк fastfetch). Никаких сторонних программ — только встроенные команды Linux.

📋 Показать полный блок с алиасами 
# ============================================
# КРУТЫЕ АЛИАСЫ ДЛЯ ВИЗУАЛИЗАЦИИ ТЕРМИНАЛА
# ============================================

# 1. Красивый вывод файлов
alias ll='ls -lah --color=auto --group-directories-first'
alias la='ls -A --color=auto'
alias l='ls -CF --color=auto'
alias lr='ls -lah --color=auto --group-directories-first -R'
alias lt='ls -lat --color=auto'

# 2. Цветной grep
alias grep='grep --color=auto'
alias egrep='egrep --color=auto'
alias fgrep='fgrep --color=auto'

# 3. Безопасные команды
alias rm='rm -i'

# 4. Системная информация
alias mem='free -h'

alias storage='du -sh * 2>/dev/null | sort -h'
alias ports='ss -tulpn 2>/dev/null'
alias processes='ps auxf --sort=-%cpu | head -20'

# 5. Сеть
alias myip='curl -s ifconfig.me && echo ""'
alias localip='ip -4 addr show | grep -oP "(?<=inet ).*(?=/)"'
alias scanports='sudo nmap -sS localhost 2>/dev/null || echo "nmap не установлен"'

# 6. Навигация
alias ..='cd ..'
alias ...='cd ../..'
alias ....='cd ../../..'
alias .2='cd ../..'
alias .3='cd ../../..'
alias -- -='cd -'

# 7. Разделители
alias line='printf "%$(tput cols)s\n"|tr " " "="'
alias line2='printf "%$(tput cols)s\n"|tr " " "-"'

# 8. Красивая шапка "sysinfo"
alias sysinfo='echo -e "\n\033[1;36m════════════════════════════════════════════════════════\033[0m"; echo -e "\033[1;32m  SYSTEM INFORMATION\033[0m"; echo -e "\033[1;36m════════════════════════════════════════════════════════\033[0m"; echo -e "\033[1;33m»\033[0m User: \033[1;37m$USER\033[0m"; echo -e "\033[1;33m»\033[0m Host: \033[1;37m$HOSTNAME\033[0m"; echo -e "\033[1;33m»\033[0m OS: \033[1;37m$(lsb_release -ds 2>/dev/null || cat /etc/os-release 2>/dev/null | grep PRETTY_NAME | cut -d= -f2 | tr -d \")\033[0m"; echo -e "\033[1;33m»\033[0m Kernel: \033[1;37m$(uname -r)\033[0m"; echo -e "\033[1;33m»\033[0m Uptime: \033[1;37m$(uptime -p | sed "s/up //")\033[0m"; echo -e "\033[1;36m════════════════════════════════════════════════════════\033[0m\n"'

# 9. Цветной промпт с веткой Git
parse_git_branch() {
    git branch 2>/dev/null | sed -e '/^[^*]/d' -e 's/* \(.*\)/(\1)/'
}
export PS1='\[\e[1;32m\]\u\[\e[0m\]@\[\e[1;36m\]\h\[\e[0m\]:\[\e[1;34m\]\W\[\e[0m\]\[\e[1;33m\]$(parse_git_branch)\[\e[0m\]\$ '

# 10. Приветствие
echo -e "\n\033[1;35m━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\033[0m"
echo -e "\033[1;36m  ДОБРО ПОЖАЛОВАТЬ, \033[1;33m$USER\033[0m"
echo -e "\033[1;35m━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\033[0m"
echo -e "\033[1;32m  Дата:\033[0m \033[1;37m$(date "+%A, %d %B %Y")\033[0m"
echo -e "\033[1;32m  Время:\033[0m \033[1;37m$(date "+%H:%M:%S")\033[0m"

nano ~/.bashrc

✅ После этого терминал станет цветным, информативным и эстетичным.
🔍 6. Проверка работоспособности

Закройте терминал и откройте заново — вы должны увидеть:

  • Приветствие с датой, временем и uptime
  • Логотип Astra Linux и системные параметры (fastfetch)
  • Цветной промпт user@host:~$
  • Алиасы (ll, sysinfo, mem) работают без ошибок
🎉 Всё готово! Ваш терминал выглядит стильно и профессионально.
❓ Частые вопросы
🔹 fastfetch: команда не найдена
Убедитесь, что алиас прописан в ~/.bashrc. Выполните source ~/.bashrc или перезапустите терминал.
🔹 Ошибка JsonConfig: Property color
В config.jsonc не должно быть устаревшего поля keyWidth — используйте "key": { "width": 15 }.
🔹 Хочу убрать автозапуск fastfetch
Закомментируйте или удалите строку fastfetch в конце ~/.bashrc.
⚠️ Для команд myip и scanports может потребоваться интернет или установка nmap / curl (необязательно).
